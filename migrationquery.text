SET FOREIGN_KEY_CHECKS=0;
-- MySQL Deep Audit Migration Scripts from `multi-tenancy` to `metis`
-- Make sure to run these scripts on a database where both schemas are accessible,
-- or export the data from `multi-tenancy` and run these transformed inserts into `metis`.
-- Assuming the source database is `central_database` and target is `metis`.

-- 0. Migrate Users
INSERT IGNORE INTO metis.users (
    id, name, email, phone, address_line_1, designation, 
    email_verified_at, password, is_active, remember_token, 
    created_at, updated_at, deleted_at
)
SELECT 
    id, name, email, phone, location, designation, 
    email_verified_at, password, 
    CASE WHEN status = 'active' THEN 1 ELSE 0 END, 
    remember_token, 
    created_at, updated_at, deleted_at
FROM central_database.users;

-- 1. Migrate Customers to Parties
INSERT IGNORE INTO metis.parties (
    id, uuid, party_code, type, firstname, lastname, 
    email, phone, alternatemobile, relative_phone, 
    source, category, company_name, gst_no, pan_no, 
    land_area, land_unit, crops, irrigation_type, 
    credit_limit, outstanding_balance, credit_valid_till, 
    aadhaar_last4, kyc_completed, kyc_verified_at, 
    first_purchase_at, last_purchase_at, orders_count, 
    status, is_active, is_blacklisted, internal_notes, tags, 
    created_by, updated_by, created_at, updated_at, deleted_at
)
SELECT 
    id, 
    uuid, 
    customer_code, 
    'customer', 
    first_name, 
    last_name, 
    email, 
    mobile, 
    phone_number_2, 
    relative_phone, 
    CASE WHEN source IS NOT NULL THEN JSON_ARRAY(source) ELSE NULL END,
    category, 
    company_name, 
    gst_number, 
    pan_number, 
    land_area, 
    land_unit, 
    crops, 
    CASE WHEN irrigation_type IS NOT NULL THEN JSON_ARRAY(irrigation_type) ELSE NULL END,
    credit_limit, 
    outstanding_balance, 
    credit_valid_till, 
    aadhaar_last4, 
    kyc_completed, 
    kyc_verified_at, 
    first_purchase_at, 
    last_purchase_at, 
    orders_count, 
    CASE WHEN is_active = 1 THEN 'active' ELSE 'inactive' END,
    is_active, 
    is_blacklisted, 
    internal_notes, 
    tags, 
    created_by, 
    updated_by, 
    created_at, 
    updated_at, 
    deleted_at
FROM central_database.customers;

-- 2. Migrate Customer Addresses to Party Addresses
INSERT IGNORE INTO metis.party_addresses (
    id, party_id, label, address_line_1, address_line_2, 
    village_name, post_office, taluka, district, state, 
    pincode, is_default, status, created_at, updated_at, deleted_at
)
SELECT 
    id, 
    customer_id, 
    IFNULL(label, 'Primary'), 
    address_line1, 
    address_line2, 
    village, 
    post_office, 
    taluka, 
    district, 
    state, 
    pincode, 
    is_default, 
    'active', 
    created_at, 
    updated_at, 
    deleted_at
FROM central_database.customer_addresses;

-- 3. Migrate Orders
INSERT IGNORE INTO metis.orders (
    id, order_no, type, party_id, order_date, 
    total_amount, tax_amount, discount_amount, coupon_code, net_amount, 
    status, warehouse_id, 
    shipping_address_id, billing_address_id, 
    pending_at, -- mapped to placed_at
    created_by, updated_by, created_at, updated_at, deleted_at
)
SELECT 
    id, 
    order_number, 
    'sale', 
    customer_id, 
    IFNULL(placed_at, created_at), -- Ensured order_date is properly set
    total_amount, 
    tax_amount, 
    discount_amount, 
    discount_code, 
    grand_total, 
    CASE 
        WHEN status = 'shipped' THEN 'dispatched'
        WHEN status = 'completed' THEN 'delivered'
        WHEN status = 'scheduled' THEN 'future_order'
        ELSE status 
    END as status, 
    warehouse_id, 
    shipping_address_id, 
    billing_address_id, 
    IFNULL(placed_at, created_at) as pending_at, 
    created_by, 
    updated_by, 
    created_at, 
    updated_at, 
    deleted_at
FROM central_database.orders;

-- 4. Migrate Order Items
INSERT IGNORE INTO metis.order_items (
    id, order_id, product_id, quantity, unit_price, 
    tax_rate, discount_amount, total_amount, created_at, updated_at
)
SELECT 
    id, 
    order_id, 
    product_id, 
    quantity, 
    unit_price, 
    tax_percent, 
    discount_amount, 
    total_price, 
    created_at, 
    updated_at
FROM central_database.order_items;

-- 5. Migrate Invoices
INSERT IGNORE INTO metis.invoices (
    id, invoice_no, order_id, invoice_date, due_date, 
    total_amount, tax_amount, net_amount, status, 
    created_at, updated_at, deleted_at
)
SELECT 
    id, 
    invoice_number, 
    order_id, 
    issue_date, 
    due_date, 
    total_amount, 
    0, -- default tax_amount, not clearly separated in multi_tenancy invoices
    total_amount, -- default net_amount
    CASE 
        WHEN status IN ('draft', 'sent', 'overdue') THEN 'unpaid'
        WHEN status = 'void' THEN 'cancelled'
        ELSE 'paid'
    END,
    created_at, 
    updated_at, 
    deleted_at
FROM central_database.invoices;

-- 6. Migrate Payments
INSERT IGNORE INTO metis.payments (
    id, payment_no, invoice_id, order_id, amount, 
    payment_method, transaction_id, payment_date, status, 
    created_at, updated_at
)
SELECT 
    id, 
    CONCAT('PAY-', id), -- Multi-tenancy lacks explicit payment_no
    invoice_id, 
    order_id, 
    amount, 
    method, 
    transaction_id, 
    paid_at, 
    'completed', -- Default mapping
    created_at, 
    updated_at
FROM central_database.payments;

-- 7. Migrate Shipments
INSERT IGNORE INTO metis.shipments (
    id, shipment_no, order_id, carrier_name, tracking_no, 
    status, shipped_at, delivered_at, estimated_delivery_date, 
    actual_weight_g, created_at, updated_at
)
SELECT 
    id, 
    CONCAT('SHP-', id), -- Multi-tenancy lacks explicit shipment_no
    order_id, 
    carrier, 
    tracking_number, 
    CASE 
        WHEN status = 'shipped' THEN 'in_transit'
        ELSE status
    END, 
    shipped_at, 
    delivered_at, 
    estimated_delivery_date, 
    weight * 1000, -- Assuming multi-tenancy was kg, metis is grams
    created_at, 
    updated_at
FROM central_database.shipments;

-- 8. Migrate Returns
INSERT IGNORE INTO metis.order_returns (
    id, order_id, return_no, status, reason, 
    refund_amount, created_at, updated_at
)
SELECT 
    id, 
    order_id, 
    rma_number, 
    CASE WHEN status = 'requested' THEN 'pending' ELSE status END,
    reason, 
    IFNULL(refunded_amount, 0),
    created_at, 
    updated_at
FROM central_database.returns;

-- 14. Migrate Products
INSERT IGNORE INTO metis.products (
    id, name, sku, slug, category_id, brand_id, barcode, image_path, 
    weight_g, purchase_price, mrp, selling_price, default_discount, 
    default_discount_type, min_stock_level, manage_stock, is_active, 
    application_instructions, status, description, 
    created_at, updated_at, deleted_at

SELECT 
    id, name, IFNULL(sku, CONCAT('SKU-', id)), slug, category_id, brand_id, barcode, image, 
    weight * 1000, IFNULL(cost_price, 0), IFNULL(mrp, 0), price, IFNULL(default_discount_value, 0), 
    IFNULL(default_discount_type, 'fixed'), reorder_level, manage_stock, is_active, 
    usage_instructions, CASE WHEN is_active = 1 THEN 'published' ELSE 'draft' END, description, 
    created_at, updated_at, deleted_at
FROM central_database.products;

-- 21. Migrate Stock Movements
INSERT IGNORE INTO metis.stock_movements (
    id, product_id, warehouse_id, reference_type, reference_id, 
    quantity, type, performed_by, created_at, updated_at
)
SELECT 
    m.id, s.product_id, s.warehouse_id, 
    CASE 
        WHEN m.type = 'purchase' THEN 'Order'
        WHEN m.type = 'sale' THEN 'Order'
        ELSE m.type 
    END, 
    m.reference_id, m.quantity, 
    CASE 
        WHEN m.type IN ('purchase', 'transfer', 'return') THEN 'in' 
        WHEN m.type = 'sale' THEN 'out'
        WHEN m.type = 'adjustment' THEN 'adjustment'
        ELSE 'adjustment'
    END,
    m.user_id, m.created_at, m.updated_at
FROM central_database.inventory_movements m
JOIN central_database.inventory_stocks s ON m.stock_id = s.id;
